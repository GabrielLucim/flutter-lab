Ver um exemplo Flutter completo do Painel de Missões

Não precisa decorar este arquivo agora. Leia procurando as peças que apareceram na trilha: MaterialApp, Scaffold, AppBar, Column, botão e setState.

import 'package:flutter/material.dart';

void main() {
  runApp(const MeuPainelApp());
}

class MeuPainelApp extends StatefulWidget {
  const MeuPainelApp({super.key});

  @override
  State<MeuPainelApp> createState() => _MeuPainelAppState();
}

class _MeuPainelAppState extends State<MeuPainelApp> {
  int feitas = 2;

  @override
  Widget build(BuildContext context) {
    return MaterialApp(
      debugShowCheckedModeBanner: false,
      home: Scaffold(
        appBar: AppBar(title: const Text('Meu Painel')),
        body: Center(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              Text('Missões feitas: $feitas'),
              const SizedBox(height: 12),
              ElevatedButton(
                onPressed: () {
                  setState(() {
                    feitas++;
                  });
                },
                child: const Text('Marcar uma como feita'),
              ),
            ],
          ),
        ),
      ),
    );
  }
}
